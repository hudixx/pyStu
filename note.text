SELECT
      u.id,
      u.name,
      t.id    AS todo_id,
      t.title
  FROM users AS u
  LEFT JOIN todos AS t                          -- 左连接：没有待办的用户也留着
      ON t.user_id = u.id
  WHERE u.created_at BETWEEN '2026-01-01' AND '2026-12-31'   -- 注册日落在这一年
      AND (                                         -- 下面整段是一个 AND 条件
            t.id IS NULL                            -- 左连接上没配到待办
            OR (                                    -- 或者配到了，并且同时满足两件
                t.priority > 1
                AND t.id IN (                       -- 子查询：紧急标签指向的那些待办
                    SELECT todo_id
                    FROM todo_tags
                    WHERE tag = 'urgent'
                )
            )
          );
